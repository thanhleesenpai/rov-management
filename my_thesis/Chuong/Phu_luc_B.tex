\hline
\textbf{Tên} & Đăng nhập bằng Email và Mật khẩu \\
\hline
\textbf{Tác nhân} & Viewer, Operator, Admin \\
\hline
\textbf{Tiền điều kiện} & Tác nhân có tài khoản đã được tạo trong hệ thống. \\
\hline
\textbf{Luồng chính} &
(1) Tác nhân truy cập trang đăng nhập, nhập địa chỉ email và mật khẩu. \newline

\hline
\textbf{Hậu điều kiện} & Tác nhân được xác thực và có thể truy cập các tính năng theo quyền hạn vai trò. \\
\hline
\end{longtable}

\begin{longtable}{|p{3.5cm}|p{10.0cm}|}

\hline
\textbf{Mã Use Case} & UC-B02 \\
\hline
\textbf{Tên} & Đăng nhập bằng tài khoản Google (OAuth 2.0) \\
\hline
\textbf{Tác nhân} & Viewer, Operator, Admin \\
\hline
\textbf{Tiền điều kiện} & Tác nhân có tài khoản Google hợp lệ. \\
\hline
\textbf{Luồng chính} &
(1) Tác nhân truy cập trang đăng nhập và chọn \textit{``Continue with Google''}. \newline
(2) Hệ thống chuyển hướng sang trang xác thực của Google (OAuth consent screen). \newline
(3) Tác nhân xác nhận cấp quyền; Google trả về thông tin hồ sơ (email, tên, avatar). \newline
(4) Hệ thống kiểm tra email trong cơ sở dữ liệu: nếu là người dùng mới, tự động tạo tài khoản với vai trò mặc định \textit{Viewer}; nếu đã tồn tại, liên kết \texttt{googleId}. \newline
(5) Hệ thống cấp phát JWT và chuyển hướng về \texttt{/auth/callback} trên frontend. \newline
(6) Tác nhân được đưa vào trang Dashboard. \\
\hline
\textbf{Luồng phát sinh} & Tác nhân hủy xác thực tại trang Google $\to$ hệ thống đưa về trang đăng nhập với thông báo lỗi; tài khoản bị vô hiệu hóa $\to$ trả lỗi 403. \\
\hline
\textbf{Hậu điều kiện} & Tác nhân đăng nhập thành công; tài khoản Google liên kết với hệ thống; tab \textit{``Đổi mật khẩu''} bị ẩn trên Profile vì Google user không có mật khẩu. \\
\hline
\end{longtable}

\section{UC-B06: Tải lên Tệp phương tiện (Media Upload)}

\begin{longtable}{|p{3.5cm}|p{10.0cm}|}
\caption{Đặc tả Use Case UC-B06: Tải lên Tệp phương tiện qua Presigned URL}\label{tab:ucb06}\\
\hline
\textbf{Mã Use Case} & UC-B06 \\
\hline
\textbf{Tên} & Tải lên Tệp phương tiện (Media Upload qua AWS S3 Presigned URL) \\
\hline
\textbf{Tác nhân} & Operator, Admin \\
\hline
\textbf{Tiền điều kiện} & Lượt lặn (Trip) đã tồn tại; tác nhân có quyền Operator/Admin. \\
\hline
\textbf{Luồng chính} &
(1) Tác nhân chọn tệp Video hoặc Ảnh trong giao diện Upload. \newline
(2) Frontend gửi \texttt{POST /media/upload-url\allowbreak{} \{filename, type, size\}} đến backend; hệ thống trả về \texttt{presignedUrl} và \texttt{mediaId}. \newline
(3) Frontend tải file trực tiếp lên AWS S3 bằng \texttt{PUT presignedUrl} (không qua backend, không tốn băng thông server). \newline
(4) Frontend gửi \texttt{POST /media/:id/confirm}; hệ thống cập nhật \texttt{media.status = 'ready'} và áp dụng cơ chế ưu tiên: nếu \texttt{recordedAt} đã được gán từ \texttt{trip\_master.json} (bước tải lên lô) thì giữ nguyên; ngược lại, tự động parse từ tên file nếu có mẫu \texttt{\_YYYYMMDD\_HHMMSS}; nếu không, \texttt{recordedAt} là \texttt{null}. \newline
(5) File xuất hiện trong MediaGallery của lượt lặn. \\
\hline
\textbf{Luồng phát sinh} & Bước PUT S3 thất bại $\to$ tác nhân không gọi confirm, document Media giữ trạng thái \textit{pending} và được dọn dẹp sau; file lớn hơn 500MB $\to$ presigned URL từ chối. \\
\hline
\textbf{Hậu điều kiện} & Tệp phương tiện được lưu trên S3 và liên kết với Trip; \texttt{analysisStatus = 'idle'} (chưa phân tích YOLO). \\
\hline
\end{longtable}
